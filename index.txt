    <!-- Script logic -->
    <script>
        let currentLang = 'hi';
        let activeTab = 'labour';

        // Sample Data
        let labours = [
            { id: 1, name: "Mahendra Singh", category: "raj mistri", phone: "9876543210", location: "Main Chowk, Tapukara" },
            { id: 2, name: "Rahul Kumar", category: "helper", phone: "9123456789", location: "Joriya Road, Tapukara" },
            { id: 3, name: "Sunil Sharma", category: "electrician", phone: "9988776655", location: "RIICO Chowk" }
        ];

        let rooms = [
            { id: 1, name: "Shyam Lal (Room Owner)", price: "₹3,500 / माह", phone: "9811122333", location: "Near Bus Stand, Tapukara (Single Room)" },
            { id: 2, name: "Gupta Ji PG", price: "₹5,000 / माह", phone: "9822233444", location: "Joriya Road (PG for Boys)" }
        ];

        function renderLabour(filter = 'all') {
            const listEl = document.getElementById('labourList');
            listEl.innerHTML = '';
            
            let filtered = labours;
            if (filter !== 'all') {
                filtered = labours.filter(l => l.category === filter);
            }

            if (filtered.length === 0) {
                listEl.innerHTML = `<p class="text-center text-gray-400 text-xs py-4">Koi labour uplabdh nahi hai.</p>`;
                return;
            }

            filtered.forEach(item => {
                listEl.innerHTML += `
                    <div class="bg-white p-3.5 rounded-xl border shadow-sm flex justify-between items-center">
                        <div>
                            <span class="text-xs bg-blue-100 text-blue-800 px-2 py-0.5 rounded-full font-medium uppercase">${item.category}</span>
                            <h3 class="font-bold text-gray-900 text-sm mt-1">${item.name}</h3>
                            <p class="text-xs text-gray-500">📍 ${item.location}</p>
                        </div>
                        <div class="flex gap-2">
                            <a href="tel:${item.phone}" class="bg-green-600 text-white p-2.5 rounded-xl shadow hover:bg-green-700 text-xs font-bold">📞 Call</a>
                            <a href="https://wa.me/91${item.phone}?text=Namaste,%20mujhe%20aapki%20zaroorat%20hai%20Tapukara%20mein." target="_blank" class="bg-emerald-500 text-white p-2.5 rounded-xl shadow hover:bg-emerald-600 text-xs font-bold">💬 WhatsApp</a>
                        </div>
                    </div>
                `;
            });
        }

        function renderRooms() {
            const listEl = document.getElementById('roomList');
            listEl.innerHTML = '';

            rooms.forEach(item => {
                listEl.innerHTML += `
                    <div class="bg-white p-3.5 rounded-xl border shadow-sm flex justify-between items-center">
                        <div>
                            <span class="text-xs bg-green-100 text-green-800 px-2 py-0.5 rounded-full font-medium">Room Available</span>
                            <h3 class="font-bold text-gray-900 text-sm mt-1">${item.name}</h3>
                            <p class="text-xs text-green-700 font-semibold">💰 ${item.price}</p>
                            <p class="text-xs text-gray-500">📍 ${item.location}</p>
                        </div>
                        <div class="flex gap-2">
                            <a href="tel:${item.phone}" class="bg-green-600 text-white p-2.5 rounded-xl shadow hover:bg-green-700 text-xs font-bold">📞 Call</a>
                            <a href="https://wa.me/91${item.phone}?text=Namaste,%20mujhe%20room%20chahiye%20Tapukara%20mein." target="_blank" class="bg-emerald-500 text-white p-2.5 rounded-xl shadow hover:bg-emerald-600 text-xs font-bold">💬 WhatsApp</a>
                        </div>
                    </div>
                `;
            });
        }

        function switchTab(tab) {
            activeTab = tab;
            document.querySelectorAll('.tab-btn').forEach(b => {
                b.classList.remove('border-b-2', 'border-blue-600', 'text-blue-600', 'font-semibold');
                b.classList.add('text-gray-500', 'font-medium');
            });

            if (tab === 'labour') {
                document.getElementById('tabLabour').classList.add('border-b-2', 'border-blue-600', 'text-blue-600', 'font-semibold');
                document.getElementById('sectionLabour').classList.remove('hidden');
                document.getElementById('sectionRooms').classList.add('hidden');
                renderLabour();
            } else {
                document.getElementById('tabRooms').classList.add('border-b-2', 'border-blue-600', 'text-blue-600', 'font-semibold');
                document.getElementById('sectionRooms').classList.remove('hidden');
                document.getElementById('sectionLabour').classList.add('hidden');
                renderRooms();
            }
        }

        function openModal(type) {
            document.getElementById('dataModal').classList.remove('hidden');
            document.getElementById('formType').value = type;
            if (type === 'labour') {
                document.getElementById('modalTitle').innerText = 'लेबर / मिस्त्री के रूप में जुड़ें';
                document.getElementById('categoryGroup').classList.remove('hidden');
                document.getElementById('priceGroup').classList.add('hidden');
                document.getElementById('labelLocation').innerText = 'टपुकड़ा में पता / चौक (Location)';
            } else {
                document.getElementById('modalTitle').innerText = 'किराए का रूम / मकान जोड़ें';
                document.getElementById('categoryGroup').classList.add('hidden');
                document.getElementById('priceGroup').classList.remove('hidden');
                document.getElementById('labelLocation').innerText = 'रूम का पूरा पता / विवरण (Address)';
            }
        }

        function closeModal() {
            document.getElementById('dataModal').classList.add('hidden');
            document.getElementById('entryForm').reset();
        }

        function handleFormSubmit(e) {
            e.preventDefault();
            const type = document.getElementById('formType').value;
            const name = document.getElementById('inputName').value;
            const phone = document.getElementById('inputPhone').value;
            const location = document.getElementById('inputLocation').value;

            if (type === 'labour') {
                const category = document.getElementById('inputCategory').value;
                labours.unshift({ id: Date.now(), name, category, phone, location });
                renderLabour();
                switchTab('labour');
            } else {
                const price = document.getElementById('inputPrice').value || "₹3,000 / माह";
                rooms.unshift({ id: Date.now(), name, price: price + " / माह", phone, location });
                renderRooms();
                switchTab('rooms');
            }
            closeModal();
            alert('सफलतापूर्वक दर्ज हो गया!');
        }

        function filterLabour(cat) {
            document.querySelectorAll('#sectionLabour .flex.gap-2 button').forEach(b => b.classList.remove('bg-blue-600', 'text-white', 'active-filter'));
            event.target.classList.add('bg-blue-600', 'text-white', 'active-filter');
            renderLabour(cat);
        }

        function toggleLanguage() {
            currentLang = currentLang === 'hi' ? 'en' : 'hi';
            document.getElementById('langBtnText').innerText = currentLang === 'hi' ? 'English' : 'हिंदी';
            document.getElementById('appTitle').innerText = currentLang === 'hi' ? '🛠️ टपुकड़ा लेबर & रूम हॉप' : '🛠️ Tapukara Labour & Room Hub';
            document.getElementById('appSubtitle').innerText = currentLang === 'hi' ? 'चौक के मजदूर, मिस्त्री और रूम की जानकारी एक जगह' : 'Daily Labour, Masons & Room Rentals in Tapukara';
        }

        // Initial Load
        renderLabour();
    </script>
